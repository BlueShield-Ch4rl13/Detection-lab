# Consentimiento concedido a una aplicacion no verificada  [high]
from * metadata _id, _index, _version | where (properties.message in ("Consent to application", "Add delegated permission grant", "Add app role assignment grant to user")) and (TargetResources like "*Mail.Read*" or TargetResources like "*Mail.ReadWrite*" or TargetResources like "*Mail.Send*" or TargetResources like "*Files.ReadWrite.All*" or TargetResources like "*Directory.ReadWrite.All*" or TargetResources like "*User.ReadWrite.All*" or TargetResources like "*offline_access*")

# Metodo MFA registrado tras un inicio de sesion de riesgo  [medium]
from * metadata _id, _index, _version | where properties.message in ("User registered security info", "Admin registered security info", "User registered all required security info", "User changed default security info")

# Fallo de autenticacion en Entra ID por credencial invalida  [informational]
from * metadata _id, _index, _version | where ResultType in ("50126", "50053", "50055")
| eval timebucket=date_trunc(1hours, @timestamp) | stats value_count=count_distinct(UserPrincipalName) by timebucket, IPAddress
| where value_count >= 15

# Inicio de sesion correcto en Entra ID  [informational]
from * metadata _id, _index, _version | where ResultType=="0"
| eval timebucket=date_trunc(30minutes, @timestamp) | stats value_count=count_distinct(AutonomousSystemNumber) by timebucket, CorrelationId
| where value_count >= 2

# Politica de acceso condicional modificada o deshabilitada  [high]

# Credencial anadida a una aplicacion o service principal  [high]

# Inicio de sesion desde una red de anonimizacion  [high]

# Dispositivo no conforme accediendo a un recurso corporativo  [medium]
